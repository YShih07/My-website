    <script>
        let scene, camera, renderer, torsoGroup;
        let mouseX = 0, mouseY = 0;
        let targetRotationX = 0, targetRotationY = 0;

        function init() {
            const container = document.getElementById('canvas-container');

            // 1. 创建 3D 场景
            scene = new THREE.Scene();

            // 2. 设置相机
            camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.z = 6.5;

            // 3. 设置渲染器
            renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            renderer.toneMapping = THREE.ACESFilmicToneMapping;
            renderer.toneMappingExposure = 1.3;
            container.appendChild(renderer.domElement);

            // 4. 浅紫色冰霜磨砂 3D 物理材质
            const frostedGlassMaterial = new THREE.MeshPhysicalMaterial({
                color: new THREE.Color('#cbb6f8'),      // 浅紫冰霜基色
                emissive: new THREE.Color('#311854'),   // 内部深紫发光
                roughness: 0.38,                        // 冰霜雾面粗糙度
                metalness: 0.05,
                transmission: 0.85,                     // 透光折射率
                thickness: 1.2,
                ior: 1.4,
                transparent: true,
                opacity: 0.92,
                clearcoat: 0.4,                         // 表面微光表层
                clearcoatRoughness: 0.2
            });

            // 5. 生成精细比例的艺术躯干雕塑
            torsoGroup = new THREE.Group();

            // 主躯干结构
            const torsoGeo = new THREE.CylinderGeometry(0.75, 0.45, 1.9, 32, 16);
            torsoGeo.scale(1.2, 1.0, 0.65);
            const mainBody = new THREE.Mesh(torsoGeo, frostedGlassMaterial);
            torsoGroup.add(mainBody);

            // 胸肌结构 (左/右)
            const pecGeo = new THREE.BoxGeometry(0.48, 0.4, 0.3);
            const pecL = new THREE.Mesh(pecGeo, frostedGlassMaterial);
            pecL.position.set(-0.28, 0.45, 0.22);
            pecL.rotation.z = -0.08;
            torsoGroup.add(pecL);

            const pecR = pecL.clone();
            pecR.position.x = 0.28;
            pecR.rotation.z = 0.08;
            torsoGroup.add(pecR);

            // 腹肌结构
            for (let i = 0; i < 3; i++) {
                const absGeo = new THREE.BoxGeometry(0.22, 0.22, 0.15);
                const absL = new THREE.Mesh(absGeo, frostedGlassMaterial);
                absL.position.set(-0.15, 0.08 - i * 0.26, 0.2);
                torsoGroup.add(absL);

                const absR = absL.clone();
                absR.position.x = 0.15;
                torsoGroup.add(absR);
            }

            // 肩部关节点
            const shoulderGeo = new THREE.SphereGeometry(0.35, 24, 24);
            shoulderGeo.scale(1.1, 0.9, 0.8);
            const shoulderL = new THREE.Mesh(shoulderGeo, frostedGlassMaterial);
            shoulderL.position.set(-0.88, 0.82, 0);
            torsoGroup.add(shoulderL);

            const shoulderR = shoulderL.clone();
            shoulderR.position.x = 0.88;
            torsoGroup.add(shoulderR);

            // 适配尺寸：缩小雕塑体积，完美置于中央
            torsoGroup.scale.set(0.68, 0.68, 0.68);
            torsoGroup.position.y = -0.15;
            scene.add(torsoGroup);

            // 6. 灯光系统 (打出紫调边缘高光)
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.9);
            scene.add(ambientLight);

            const mainLight = new THREE.DirectionalLight(0xe8ddff, 2.8);
            mainLight.position.set(5, 8, 5);
            scene.add(mainLight);

            const purpleRimLight = new THREE.PointLight(0xa855f7, 4.5, 12);
            purpleRimLight.position.set(-4, -2, -2);
            scene.add(purpleRimLight);

            const blueRimLight = new THREE.PointLight(0x38bdf8, 3.5, 12);
            blueRimLight.position.set(4, 3, -2);
            scene.add(blueRimLight);

            // 事件绑定
            document.addEventListener('mousemove', onMouseMove);
            window.addEventListener('resize', onResize);
            animate();
        }

        function onMouseMove(event) {
            mouseX = (event.clientX - window.innerWidth / 2) / (window.innerWidth / 2);
            mouseY = (event.clientY - window.innerHeight / 2) / (window.innerHeight / 2);
        }

        function onResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function animate() {
            requestAnimationFrame(animate);

            if (torsoGroup) {
                // 平滑随鼠标旋转
                targetRotationY = mouseX * 0.5;
                targetRotationX = mouseY * 0.3;

                torsoGroup.rotation.y += (targetRotationY - torsoGroup.rotation.y) * 0.05;
                torsoGroup.rotation.x += (targetRotationX - torsoGroup.rotation.x) * 0.05;
                
                // 缓动悬浮呼吸效果
                torsoGroup.position.y = -0.15 + Math.sin(Date.now() * 0.0015) * 0.06;
            }

            renderer.render(scene, camera);
        }

        init();
    </script>
